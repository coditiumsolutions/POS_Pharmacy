# Version: 1.1 | UpdatedAt: 21/11/2026
# Database documentation for POS Pharmacy
# Source: live schema introspection (credentials not stored)
# Note: PK columns use uid (maps to category_uid / product_uid / inventory_uid)
# Note: categories table replaces product_categories

FIELD MAPPING (Pharmacy schema)
-------------------------------
categories.uid = category_uid
categories.category_name = category_name
products.uid = product_uid
products.barcode = barcode
products.product_name = product_name / ItemName
products.generic_name = generic_name
products.company_name = company_name
products.strength = strength
products.pack_unit = pack_unit
products.units_per_pack = units_per_pack
products.sale_price = sale_price / SaleValue
products.discount_percent = discount_percent
inventories.uid = inventory_uid
inventories.batch_no = batch_no / Batch
inventories.expiry_date = expiry_date / Expiry
inventories.quantity_in_stock = Quantity
inventories.sale_price = SalePrice (batch-level)
inventories.remarks = Remarks
purchase_invoices.uid = purchase_invoice_uid
purchase_invoices.invoice_no = invoice_no
purchase_invoices.supplier_name = supplier_name
purchase_invoice_details.uid = purchase_detail_uid
purchase_invoice_details.inventory_uid = links line to batch inventory
purchase_invoice_details.received_base_quantity = (qty * units_per_pack) + bonus + loose
